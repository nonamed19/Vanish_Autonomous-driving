package com.moonshot.member_service.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;
import org.springframework.security.config.annotation.web.builders.HttpSecurity;
import org.springframework.security.config.annotation.web.configuration.EnableWebSecurity;
import org.springframework.security.crypto.bcrypt.BCryptPasswordEncoder;
import org.springframework.security.web.SecurityFilterChain;

@Configuration
@EnableWebSecurity
public class SecurityConfig {

    @Bean
    public SecurityFilterChain securityFilterChain(HttpSecurity http) throws Exception {
        http
                .csrf((csrfConfig) ->
                        csrfConfig.disable()
                )
                .authorizeHttpRequests(auth -> auth.anyRequest().permitAll()); // 모든 요청 허용

        return http.build();
    }

    @Bean
    public BCryptPasswordEncoder passwordEncoder() {
        return new BCryptPasswordEncoder();
    }
}

import com.moonshot.member_service.dto.*;

@Slf4j
@Controller
@RequestMapping("/api/v1/auth")
public class AuthController {

    private final AuthService authService;
    private final RefreshTokenService refreshTokenService;
    private final JWTUtil jwtUtil;
    private final EmailVerificationService emailVerificationService;
    private final BCryptPasswordEncoder passwordEncoder;

    public AuthController(AuthService authService, RefreshTokenService refreshTokenService, JWTUtil jwtUtil, EmailVerificationService emailVerificationService, BCryptPasswordEncoder passwordEncoder) {
        this.authService = authService;
        this.refreshTokenService = refreshTokenService;
        this.jwtUtil = jwtUtil;
        this.emailVerificationService = emailVerificationService;
        this.passwordEncoder = passwordEncoder;
    }

    /**
     * 회원가입 API
     *
     * 사용자가 회원가입을 요청하면, 이메일, 이름, 생년월일, 비밀번호를 입력받아 회원을 등록한다.
     *
     * @param MemberSignupRequest 회원가입 요청 DTO (이메일, 이름, 생년월일, 비밀번호 포함)
     * @return 성공 시 회원가입 완료 메시지
     */
    @PostMapping("/signup")
    public ResponseEntity<String> signup(@Valid @RequestBody MemberSignupRequest memberSignupRequest) {
        try {
            // 회원가입 처리
            authService.signup(memberSignupRequest);
            return ResponseEntity.ok("회원가입 성공");
        } catch (IllegalArgumentException e) {
            // 이메일 중복 오류 처리
            return ResponseEntity.badRequest().body(e.getMessage());
        }
    }

    /**
     * 로그인 API
     * 사용자가 이메일과 비밀번호를 전송하면, 인증 후 JWT 토큰을 반환한다.
     */
    @PostMapping("/login")
    public ResponseEntity<LoginResponse> login(@Valid @RequestBody MemberLoginRequest loginRequest) {
        try {
            LoginResponse response = authService.login(loginRequest);
            return ResponseEntity.ok(response);
        } catch (IllegalArgumentException e) {
            log.error("로그인 실패: {}", e.getMessage());
            return ResponseEntity.status(HttpStatus.UNAUTHORIZED).body(null);
        } catch (Exception e) {
            log.error("로그인 중 오류 발생: {}", e.getMessage());
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body(null);
        }
    }

    @PostMapping("/logout")
    public ResponseEntity<String> logout(@RequestHeader("Authorization") String token) {
        String refreshToken = jwtUtil.resolveToken(token);

        refreshTokenService.verifyRefreshToken(refreshToken)
                .ifPresent(rt -> refreshTokenService.deleteByUser(rt.getUser())); // 삭제

        return ResponseEntity.ok("로그아웃 완료");

    public ResponseEntity<String> verifyEmailCode(@Valid @RequestBody EmailVerificationCheckRequest request) {

        if (isDuplicate) {

        }

        }
    }
}

    private final JWTUtil jwtUtil;
    public MemberController(MemberService memberService, JWTUtil jwtUtil) {
        this.memberService = memberService;
        this.jwtUtil = jwtUtil;
    }

    @PostMapping("/change-password")
    public ResponseEntity<String> changePassword(@Valid @RequestBody PasswordChangeRequest request) {
        try {
            memberService.changePassword(request);
            return ResponseEntity.ok("비밀번호 변경 성공");
        } catch (IllegalArgumentException e) {
            return ResponseEntity.badRequest().body(e.getMessage());
        } catch (Exception e) {
            log.error("비밀번호 변경 중 오류 발생: {}", e.getMessage());
            return ResponseEntity.status(HttpStatus.INTERNAL_SERVER_ERROR).body("비밀번호 변경 실패");
        }
    }

    /**
     * 회원 정보 조회 API
     *
     * 요청 헤더의 Access Token에서 회원 ID를 추출하여 해당 회원의 정보를 조회한 후,
     * 민감 정보를 제외한 응답 DTO를 반환한다.
     *
     * @param authorization "Bearer {accessToken}" 형식의 Authorization 헤더
     * @return 회원 정보 응답(MemberResponse)
     */
    @GetMapping("/profile")
    public ResponseEntity<MemberResponse> getProfile(@RequestHeader("Authorization") String authorization) {
        // "Bearer " 접두사 제거
        String token = jwtUtil.resolveToken(authorization);
        log.info("Authorization header: '{}'", authorization);
        if (token == null) {
            log.info("token is null");
            return ResponseEntity.badRequest().build();
        }
        // Access Token에서 회원 ID 추출 (false: access token 처리)
        Long memberId = jwtUtil.extractUserNo(token, false);
        MemberResponse response = memberService.getMemberProfile(memberId);
        return ResponseEntity.ok(response);
    }

    /**
     * 회원 탈퇴 API
     *
     * 클라이언트는 탈퇴 요청 시, Authorization 헤더에 Access Token과 함께,
     * 요청 본문에 탈퇴 확인용 비밀번호를 전송한다.
     *
     * @param authorization "Bearer {accessToken}" 형식의 헤더
     * @param request       WithdrawalRequest DTO (비밀번호 포함)
     * @return 탈퇴 성공 시 메시지, 실패 시 오류 메시지
     */
    @PostMapping("/withdraw")
    public ResponseEntity<String> withdraw(
            @RequestHeader("Authorization") String authorization,
            @Valid @RequestBody WithdrawalRequest request) {
        try {
            memberService.withdraw(authorization, request.getPassword());
            return ResponseEntity.ok("회원 탈퇴 완료");
        } catch (IllegalArgumentException e) {
            return ResponseEntity.badRequest().body(e.getMessage());
        }
    }
}

     * @param rawPassword 사용자가 입력한 원본 비밀번호
     * @param encoder     BCryptPasswordEncoder
     * @return 일치하면 true, 아니면 false

        // 이메일로 회원 조회
        Optional<Member> memberOpt = authRepository.findByEmail(loginRequest.getEmail());
        if (memberOpt.isEmpty()) {
            throw new IllegalArgumentException("회원이 존재하지 않습니다.");
        }
        Member member = memberOpt.get();

        // 비밀번호 검증 (Member 엔티티의 checkPassword 메서드를 사용)
        if (!member.checkPassword(loginRequest.getPassword(), passwordEncoder)) {
            throw new IllegalArgumentException("인증에 실패했습니다.");
        }

        // 토큰 생성
        String accessToken = jwtUtil.createAccessToken(member.getId(), member.getName());
        
        // Refresh Token 업데이트 (한 번 생성한 후 DB에 저장)
        RefreshToken refreshTokenEntity = refreshTokenService.updateRefreshToken(member);
        String refreshToken = refreshTokenEntity.getToken();

        return new LoginResponse(accessToken, refreshToken);
    }

     * 입력된 이메일과 생년월일로 회원을 조회한 후, 임시 비밀번호를 생성하고
     * 해당 임시 비밀번호를 BCrypt로 암호화하여 회원 비밀번호를 업데이트하고,

     * @param request 비밀번호 찾기 요청 DTO

    public void retrievePassword(PasswordRetrievalRequest request) {

        String tempPassword = generateTemporaryPassword();

        // 임시 비밀번호를 BCrypt로 암호화하여 업데이트
        String encodedTempPassword = passwordEncoder.encode(tempPassword);
        member.setPassword(encodedTempPassword);
        authRepository.save(member);

        // 임시 비밀번호를 사용자 이메일로 전송
        emailService.sendPasswordRetrievalEmail(member.getEmail(), tempPassword);

     *
     * @return 임시 비밀번호 문자열

    private String generateTemporaryPassword() {

        verification.setExpiryTime(expiryTime);
        verification.setVerified(false);

        emailVerificationRepository.save(verification);

        // 이메일 발송
        emailService.sendVerificationEmail(email, code);
    }

     * @return 인증 성공 여부 (true/false)

    public boolean verifyEmailCode(String email, String code) {

        EmailVerification verification = emailVerificationRepository.findByEmail(email)
                .orElse(null);

        }

public class RefreshTokenService {

    private final JWTUtil jwtUtil; // JWT 생성/검증 유틸

    // Refresh Token 저장 (Setter 제거, 빌더 패턴 사용)
    public RefreshToken createRefreshToken(Member user) {

                .user(user)
                .token(jwtUtil.generateRefreshToken(user)) // Refresh Token 생성

        LocalDateTime newExpiryDate = LocalDateTime.now().plusDays(7);

        Optional<RefreshToken> existingToken = refreshTokenRepository.findByUser(user);

        if (existingToken.isPresent()) {
            RefreshToken updatedToken = existingToken.get().updateToken(newToken, newExpiryDate);
            return refreshTokenRepository.save(updatedToken);
        }

        return createRefreshToken(user);
    }

    // Refresh Token 삭제 (로그아웃 시)
    public void deleteByUser(Member user) {
        refreshTokenRepository.deleteByUser(user);

import java.security.PublicKey;
import java.security.spec.PKCS8EncodedKeySpec;
import java.security.spec.X509EncodedKeySpec;
import java.time.Duration;
import java.time.Instant;
import java.time.ZonedDateTime;
import java.util.Base64;
import java.util.Date;

import com.moonshot.member_service.entity.Member;
import com.moonshot.member_service.exception.JwtValidationException;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.io.ClassPathResource;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;

import io.jsonwebtoken.Claims;
import io.jsonwebtoken.ExpiredJwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.MalformedJwtException;
import io.jsonwebtoken.UnsupportedJwtException;
import jakarta.annotation.PostConstruct;

@Slf4j
@Component
public class JWTUtil {

    /**
     * [경고]
     *
     * 비대칭키를 이용하여 인증 서버의 키는 개인키(Private Key)를 사용.
     * private_key.pem은 resources에 위치하지만 예시로 설정한 것.
     * 절대로 key 파일은 프로젝트 내부에 위치하지 않음.
     *  --> jar, war 등으로 추출 시 key 파일이 포함됨.
     *  --> Git과 같은 저장소에도 올라갈 수 있음.
     *
     */
    @Value("${jwt.private-key-path}")
    private String privateKeyPath;
    private PrivateKey privateKey;

    @Value("${jwt.public-key-path}")
    private String publicKeyPath;
    private PublicKey publicKey;

    @Value("${jwt.access-token-validity}")
    private long accessTokenValidityTime;

    @Value("${jwt.refresh-token-validity}")
    private long refreshTokenValidityTime;

    private Date maxAge;

    @PostConstruct // Bean이 생성되고, Dependency Injection이 완료된 후 초기화 해야함을 명시.
    public void init() throws Exception {
        this.privateKey = loadPrivateKey();
        this.publicKey = loadPublicKey();
    }

    public PrivateKey loadPrivateKey() throws Exception {

        ClassPathResource resource = new ClassPathResource(privateKeyPath);

        String privateKeyPEM = new String(resource.getInputStream().readAllBytes())
                .replace("-----BEGIN PRIVATE KEY-----", "")
                .replace("-----END PRIVATE KEY-----", "")
                .replaceAll("\\s", "");

        byte[] encoded = Base64.getDecoder().decode(privateKeyPEM);
        KeyFactory keyFactory = KeyFactory.getInstance("RSA");
        PKCS8EncodedKeySpec keySpec = new PKCS8EncodedKeySpec(encoded);

        return keyFactory.generatePrivate(keySpec);
    }

    private PublicKey loadPublicKey() throws Exception {
        ClassPathResource resource = new ClassPathResource(publicKeyPath);

        String publicKeyPEM = new String(resource.getInputStream().readAllBytes())
                .replace("-----BEGIN PUBLIC KEY-----", "")
                .replace("-----END PUBLIC KEY-----", "")
                .replaceAll("\\s", "");

        byte[] encoded = Base64.getDecoder().decode(publicKeyPEM);
        KeyFactory keyFactory = KeyFactory.getInstance("RSA");
        X509EncodedKeySpec keySpec = new X509EncodedKeySpec(encoded);

        return keyFactory.generatePublic(keySpec);
    }

    // Access Token 생성
    public String createAccessToken(Long userNo, String teamName) {

        Instant now = ZonedDateTime.now().toInstant(); // 토큰 생성 시간.
        Instant validity = now.plus(Duration.ofMinutes(accessTokenValidityTime)); // 토큰 유효 시간

        maxAge = Date.from(validity);

        return Jwts.builder()
                .subject(String.valueOf(userNo)) // 토큰의 식별자
                .claim("TeamName", teamName) // 토큰에 포함될 정보
                .issuedAt(Date.from(now)) // 토큰 발급 시간
                .expiration(Date.from(validity)) // 토큰 만료 시간
                .signWith(privateKey, Jwts.SIG.RS256) // 토큰 서명 (암호화) 및 방식, 비공개 키(개인키, private key)로 싸이닝.
                .compact(); // 문자열 변환
    }

    // Refresh Token 생성
    public String createRefreshToken(Long userNo) {

        Instant now = ZonedDateTime.now().toInstant(); // 토큰 생성 시간.
        Instant validity = now.plus(Duration.ofHours(refreshTokenValidityTime)); // 토큰 유효 시간

        maxAge = Date.from(validity);

        return Jwts.builder()
                .subject(String.valueOf(userNo)) // 토큰의 식별자
                .claim("TeamName", "") // 토큰에 포함될 정보
                .issuedAt(Date.from(now)) // 토큰 발급 시간
                .expiration(Date.from(validity)) // 토큰 만료 시간
                .signWith(privateKey, Jwts.SIG.RS256) // 토큰 서명 (암호화) 및 방식, 비공개 키(개인키, private key)로 싸이닝.
                .compact(); // 문자열 변환
    }

    public boolean validateToken(String token) throws JwtValidationException {
        try {
            Jwts.parser()
                    .verifyWith(publicKey) // 공개키로 복호화
                    .build()
                    .parseSignedClaims(token);
            return true;
        } catch (SecurityException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.UNAUTHORIZED_TOKEN, e);
        } catch (MalformedJwtException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.MALFORMED_TOKEN, e);
        } catch (UnsupportedJwtException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.UNSUPPORTED_TOKEN, e);
        } catch (IllegalArgumentException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.INVALID_TOKEN, e);
        } catch (ExpiredJwtException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.EXPIRED_TOKEN, e);
        }
    }

    public Claims extractAllClaims(String token, boolean isRefreshToken) {
        String actualToken = null;
        try {
            log.info("token: '{}'", token);
            log.info("isRefreshToken= '{}'", isRefreshToken);
            // refreshToken인 경우 그대로 사용, accessToken인 경우 Bearer 제거
            actualToken = isRefreshToken ? token : resolveToken(token);
            log.info("token: '{}'", actualToken);
            return Jwts.parser()
                    .clock(() -> Date.from(Instant.now()))
                    .verifyWith(publicKey)
                    .build()
                    .parseSignedClaims(actualToken)
                    .getPayload();

        } catch (ExpiredJwtException e) {
            throw new JwtValidationException(
                    JwtValidationException.JwtErrorType.EXPIRED_TOKEN,
                    e
            );
        }
    }

    public Long extractUserNo(String token, boolean isRefreshToken) {
        return Long.parseLong(extractAllClaims(token, isRefreshToken).getSubject());
    }

    public String extractUserTeamName(String token, boolean isRefreshToken) {
        return extractAllClaims(token, isRefreshToken).get("TeamName", String.class);
    }

    public Date extractExpiration(String token, boolean isRefreshToken) {
        return extractAllClaims(token, isRefreshToken).getExpiration();
    }

    // Bearer 토큰에서 JWT 추출
    public String resolveToken(String bearerToken) {
        log.info("bearerToken: '{}'", bearerToken);

        // Bearer 접두사가 없는 경우도 처리
        if (StringUtils.hasText(bearerToken)) {
            if (bearerToken.startsWith("Bearer ")) {
                return bearerToken.substring(7).trim();
            } else {
                // Bearer 접두사 없이 토큰이 직접 전달된 경우
                return bearerToken.trim();
            }
        }
        return null;
    }

    // 남은 만료 시간 계산 (밀리초)
    public Date getExpirationTime(String token, boolean isRefreshToken) {
        return extractExpiration(token, isRefreshToken);
    }

    // Refresh Token 생성
    public String generateRefreshToken(Member user) {
        Instant now = ZonedDateTime.now().toInstant(); // 토큰 생성 시간
        Instant validity = now.plus(Duration.ofHours(refreshTokenValidityTime)); // 토큰 유효 시간

        Date expiration = Date.from(validity);

        return Jwts.builder()
                .subject(String.valueOf(user.getId())) // 토큰의 식별자 (유저 ID)
                .claim("TeamName", "") // 토큰에 포함될 정보 (여기서는 빈 값으로 설정)
                .issuedAt(Date.from(now)) // 토큰 발급 시간
                .expiration(expiration) // 토큰 만료 시간
                .signWith(privateKey, Jwts.SIG.RS256) // 서명 (비공개 키 사용)
                .compact(); // 문자열 변환
    }

    public Date getMaxAge() {
        return maxAge;
    }
}

        }

        public String getMessage() {
            return message;
        }
    }

    public JwtValidationException(JwtErrorType errorType) {
        super(errorType.getMessage());
        this.errorType = errorType;
    }

    public JwtValidationException(JwtErrorType errorType, Throwable cause) {
        super(errorType.getMessage(), cause);
        this.errorType = errorType;
    }

    public JwtErrorType getErrorType() {
        return errorType;
    }
}

package com.moonshot.notification_service.security;

import com.moonshot.notification_service.exception.JwtValidationException;
import io.jsonwebtoken.Claims;
import io.jsonwebtoken.ExpiredJwtException;
import io.jsonwebtoken.Jwts;
import io.jsonwebtoken.MalformedJwtException;
import io.jsonwebtoken.UnsupportedJwtException;

import org.springframework.beans.factory.annotation.Value;

import org.springframework.util.StringUtils;

import java.security.PublicKey;
import java.security.spec.X509EncodedKeySpec;
import java.time.Instant;

@Component
public class JwtTokenProvider {

    private String publicKeyPath;  // public key가 위치한 경로 (예: "keys/public_key.pem")

    public void init() throws Exception {
        // resources에 위치한 public key 파일에서 공개키를 로드
        ClassPathResource resource = new ClassPathResource(publicKeyPath);
        String publicKeyPEM = new String(resource.getInputStream().readAllBytes())
                .replace("-----BEGIN PUBLIC KEY-----", "")
                .replace("-----END PUBLIC KEY-----", "")
                .replaceAll("\\s", "");
        byte[] encoded = Base64.getDecoder().decode(publicKeyPEM);
        KeyFactory keyFactory = KeyFactory.getInstance("RSA");
        X509EncodedKeySpec keySpec = new X509EncodedKeySpec(encoded);
        publicKey = keyFactory.generatePublic(keySpec);
    }

    public boolean validateToken(String token) throws JwtValidationException {
        try {
            // 공개키로 토큰의 서명을 검증
            Jwts.parser()
                    .verifyWith(publicKey)
                    .build()
                    .parseSignedClaims(token);
            return true;
        } catch (SecurityException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.UNAUTHORIZED_TOKEN, e);
        } catch (MalformedJwtException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.MALFORMED_TOKEN, e);
        } catch (UnsupportedJwtException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.UNSUPPORTED_TOKEN, e);
        } catch (IllegalArgumentException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.INVALID_TOKEN, e);
        } catch (ExpiredJwtException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.EXPIRED_TOKEN, e);
        }
    }

    public Claims extractAllClaims(String token, boolean isRefreshToken) {
        String actualToken = isRefreshToken ? token : resolveToken(token);
        try {
            return Jwts.parser()
                    .clock(() -> Date.from(Instant.now()))
                    .verifyWith(publicKey)
                    .build()
                    .parseSignedClaims(actualToken)
                    .getPayload();
        } catch (ExpiredJwtException e) {
            throw new JwtValidationException(JwtValidationException.JwtErrorType.EXPIRED_TOKEN, e);
        }
    }

    public Long extractUserNo(String token, boolean isRefreshToken) {
        // 토큰의 subject로 저장된 사용자 번호를 Long 타입으로 변환
        return Long.parseLong(extractAllClaims(token, isRefreshToken).getSubject());
    }

    // Bearer 토큰에서 JWT 추출
    public String resolveToken(String bearerToken) {
        // Bearer 접두사가 없는 경우도 처리
        if (StringUtils.hasText(bearerToken)) {
            if (bearerToken.startsWith("Bearer ")) {
                return bearerToken.substring(7).trim();
            } else {
                // Bearer 접두사 없이 토큰이 직접 전달된 경우
                return bearerToken.trim();
            }
        }
        return null;
    }

    // JWT 토큰과 FCM Token Request를 받아서 처리하는 메서드
    public void processFcmToken(String bearerToken, FcmTokenRequest request) {
        String token = extractToken(bearerToken);
        if (token != null && tokenProvider.validateToken(token)) {
            Long userId = tokenProvider.extractUserNo(token, false);
            saveNotification(userId, request);
        } else {
            // 토큰이 유효하지 않을 경우 예외 처리
            throw new RuntimeException("Invalid or missing JWT token");
        }
    }

    // Bearer 접두사 제거
    private String extractToken(String bearerToken) {
        if (bearerToken != null && bearerToken.startsWith("Bearer ")) {
            return bearerToken.substring(7).trim();
        }
        return bearerToken;
    }

    // 추출한 사용자 ID와 FCM Token을 DB에 저장
    private void saveNotification(Long userId, FcmTokenRequest request) {
        // 해당 사용자에 대해 이미 등록된 토큰 목록 조회
        List<FcmToken> tokens = fcmTokenRepository.findByUserId(userId);

        // 동일한 토큰이 존재하는지 확인
        for (FcmToken token : tokens) {
            if (token.getToken().equals(request.getFcmToken())) {
                // 이미 존재하면 저장하지 않고 메서드를 종료합니다.
                return;
            }
        }

        // 동일한 토큰이 없으면 새로 저장
        FcmToken notification = new FcmToken();
        notification.setUserId(userId);
        notification.setToken(request.getFcmToken());
        fcmTokenRepository.save(notification);
    }

    /**
     * 헤더에 포함된 JWT를 사용해 사용자 ID를 추출하고, 해당 사용자의 FCM 토큰으로 푸쉬 알림을 전송합니다.
     */
    public String sendPushNotification(String bearerToken, NotificationRequestDto requestDto) {
        String token = extractToken(bearerToken);
        if (token == null || !tokenProvider.validateToken(token)) {
            throw new RuntimeException("Invalid or missing JWT token");
        }
        Long userId = tokenProvider.extractUserNo(token, false);

        // DB에서 해당 사용자의 FCM 토큰 조회
        List<FcmToken> tokens = fcmTokenRepository.findByUserId(userId);
        if (tokens.isEmpty()) {
            throw new RuntimeException("User id " + userId + "에 대한 FCM 토큰을 찾을 수 없습니다.");
        }

        // 각 토큰에 대해 개별 메시지 전송
        for (FcmToken fcmToken : tokens) {
            Message message = Message.builder()
                    .setToken(fcmToken.getToken())
                    .setNotification(Notification.builder()
                            .setTitle(requestDto.getTitle())
                            .setBody(requestDto.getBody())
                            .build())
                    .build();
            try {
                String response = FirebaseMessaging.getInstance().send(message);
                System.out.println("FCM 메시지 전송 성공 - Token: " + fcmToken.getToken() + ", Response: " + response);
            } catch (Exception e) {
                System.err.println("토큰 " + fcmToken.getToken() + "에 알림 전송 실패: " + e.getMessage());
            }
        }
        return "모든 토큰으로 알림 전송 완료";
    }


    public void sendParkingCompleteNotification(String vehicleVin, String parkingLotId) {

        List<FcmToken> tokens = fcmTokenRepository.findByUserId(ownerId);

    public PaymentSummaryResponse completePayment(Long userId) {
        PaymentHistory paymentHistory = paymentHistoryRepository.findByUserIdAndStatus(userId, PaymentHistory.PaymentStatus.valueOf("P"))
                .orElseThrow(() -> new IllegalArgumentException("결제 내역이 존재하지 않습니다."));
        LocalDateTime parkingEnd = LocalDateTime.now();
        log.info("[parkingEnd]: {}", parkingEnd);
        paymentHistory.setParkingEnd(parkingEnd);

        LocalDateTime parkingStartUTC = paymentHistory.getParkingStart();
        LocalDateTime parkingEndUTC = paymentHistory.getParkingEnd();
        log.info("[결제 로직] DB에서 조회한 parkingStart(UTC 가정): {}", parkingStartUTC);
        log.info("[결제 로직] DB에서 조회한 parkingEnd(UTC 가정): {}", parkingEndUTC);

        if(parkingStartUTC == null){
            throw new IllegalStateException("입차 시각이 등록되지 않아 결제를 완료할 수 없습니다.");
        }

        ZonedDateTime zoneStart = parkingStartUTC.atZone(ZoneId.of("UTC"));
        ZonedDateTime zoneEnd = parkingEndUTC.atZone(ZoneId.of("UTC"));
        log.info("[결제 로직] ZonedS UTC 시간: {}", zoneStart);
        log.info("[결제 로직] ZonedE UTC 시간: {}", zoneEnd);
        LocalDateTime parkingStartKst = zoneStart.withZoneSameInstant(ZoneId.of("Asia/Seoul")).toLocalDateTime();
        LocalDateTime parkingEndKst = zoneEnd.withZoneSameInstant(ZoneId.of("Asia/Seoul")).toLocalDateTime();
        log.info("[결제 로직] 변환된 KST 기준 parkingStart: {}", parkingStartKst);
        log.info("[결제 로직] 변환된 KST 기준 parkingEnd: {}", parkingEndKst);

        // 총 주차 시간 계산
        long durationMinutes = Duration.between(parkingStartKst, parkingEndKst).toMinutes();
        paymentHistory.setParkingDuration((int)durationMinutes);

        // 결제 요금 계산
        BigDecimal totalFee = paymentHistory.getBaseFee();
        if(durationMinutes > 30) {
            long extraMinutes = durationMinutes - 30;
            long extraUnits = (extraMinutes + 9) / 10;// 올림 처리
            BigDecimal additionalFee = paymentHistory.getRateFee().multiply(BigDecimal.valueOf(extraUnits));
            totalFee = totalFee.add(additionalFee);
        }
        paymentHistory.setPrice(totalFee);

        paymentHistoryRepository.save(paymentHistory);

        // 클라이언트에 반환할 결제 요약 정보
        PaymentSummaryResponse summary = PaymentSummaryResponse.builder()
                .parkingLotName(paymentHistory.getOrderName())
                .parkingStart(parkingStartKst)
                .parkingEnd(parkingEndKst)
                .parkingDuration(paymentHistory.getParkingDuration())
                .totalFee(totalFee)
                .baseFee(paymentHistory.getBaseFee())
                .rateFee(paymentHistory.getRateFee())
                .build();
        return summary;
    }

package com.moonshot.payment_service.util;

import com.moonshot.payment_service.exception.JwtValidationException;
import io.jsonwebtoken.Claims;
import io.jsonwebtoken.ExpiredJwtException;
import io.jsonwebtoken.Jwts;
import jakarta.annotation.PostConstruct;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.io.ClassPathResource;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;

import java.security.KeyFactory;
import java.security.PublicKey;
import java.security.spec.X509EncodedKeySpec;
import java.time.Instant;
import java.util.Base64;
import java.util.Date;

@Slf4j
@Component
public class JWTUtil {

    @Value("${jwt.public-key-path}")
    private String publicKeyPath;
    private PublicKey publicKey;

    @PostConstruct // Bean이 생성되고, Dependency Injection이 완료된 후 초기화 해야함을 명시.
    public void init() throws Exception {
        this.publicKey = loadPublicKey();
    }

    private PublicKey loadPublicKey() throws Exception {
        ClassPathResource resource = new ClassPathResource(publicKeyPath);

        String publicKeyPEM = new String(resource.getInputStream().readAllBytes())
                .replace("-----BEGIN PUBLIC KEY-----", "")
                .replace("-----END PUBLIC KEY-----", "")
                .replaceAll("\\s", "");

        byte[] encoded = Base64.getDecoder().decode(publicKeyPEM);
        KeyFactory keyFactory = KeyFactory.getInstance("RSA");
        X509EncodedKeySpec keySpec = new X509EncodedKeySpec(encoded);

        return keyFactory.generatePublic(keySpec);
    }

    public Long extractUserNo(String token, boolean isRefreshToken) {
        return Long.parseLong(extractAllClaims(token, isRefreshToken).getSubject());
    }

    public Claims extractAllClaims(String token, boolean isRefreshToken) {
        String actualToken = null;
        try {
            log.info("token: '{}'", token);
            log.info("isRefreshToken= '{}'", isRefreshToken);
            // refreshToken인 경우 그대로 사용, accessToken인 경우 Bearer 제거
            actualToken = isRefreshToken ? token : resolveToken(token);
            log.info("token: '{}'", actualToken);
            return Jwts.parser()
                    .clock(() -> Date.from(Instant.now()))
                    .verifyWith(publicKey)
                    .build()
                    .parseSignedClaims(actualToken)
                    .getPayload();

        } catch (ExpiredJwtException e) {
            throw new JwtValidationException(
                    JwtValidationException.JwtErrorType.EXPIRED_TOKEN,
                    e
            );
        }
    }

    // Bearer 토큰에서 JWT 추출
    public String resolveToken(String bearerToken) {
        log.info("bearerToken: '{}'", bearerToken);

        // Bearer 접두사가 없는 경우도 처리
        if (StringUtils.hasText(bearerToken)) {
            if (bearerToken.startsWith("Bearer ")) {
                return bearerToken.substring(7).trim();
            } else {
                // Bearer 접두사 없이 토큰이 직접 전달된 경우
                return bearerToken.trim();
            }
        }
        return null;
    }
}

package com.moonshot.remote_service.listener;

import org.springframework.amqp.rabbit.annotation.RabbitListener; import org.springframework.stereotype.Component;

@Component public class RemoteDestinationListener {
    // remote.destination.queue로 들어오는 메시지를 처리하는 리스너
// @RabbitListener로 처리한 결과 값은 자동으로 요청쪽 convertSendAndReceive의 응답으로 전달됩니다.
    @RabbitListener(queues = "remote.destination.reply.queue")
    public String handleRemoteDestination(String message) {
        try {
            String[] parts = message.split("\\|");
            if (parts.length != 4) {
                return "error|잘못된 요청 형식입니다.";
            }
            if (!"destination".equals(parts[0])) {
                return "error|올바르지 않은 명령입니다.";
            }
            // parts[1]: 위도, parts[2]: 경도, parts[3]: 예상 도착 시간(또는 예상 소요 시간)
            return String.format("%s|%s|%s", parts[1], parts[2], parts[3]);
        } catch (Exception e) {
            return "error|" + e.getMessage();
        }
    }

}

                    requestDto.getCurrentLocation().getLatitude(),
                    requestDto.getCurrentLocation().getLongitude(),
                    timeStr);
        }
        Object response = rabbitTemplate.convertSendAndReceive(remoteExchange, goalRoutingKey, message);

        if (response != null) {
            // 반환된 response가 byte[]인 경우 이를 문자열로 변환
            String responseStr;
            if (response instanceof byte[]) {
                responseStr = new String((byte[])response, StandardCharsets.UTF_8);
            } else {
                responseStr = response.toString();
            }
            log.info("Received response string: {}", responseStr);

            String[] parts = responseStr.split("\\|");
            log.info("Split response into {} parts: {}", parts.length, java.util.Arrays.toString(parts));

            if (parts.length == 3) {
                try {
                    Double currentLat = Double.parseDouble(parts[0]);
                    Double currentLon = Double.parseDouble(parts[1]);
                    String eta = parts[2];
                    VehicleCurrentLocationDto vehicleCurrentLocation =
                            new VehicleCurrentLocationDto(currentLat, currentLon, eta);
                    return new RemoteVehicleGoalResponseDto(true, "원격 호출 목적지 설정 성공", vehicleCurrentLocation);
                } catch (NumberFormatException e) {
                    log.error("응답 메시지 파싱 중 숫자 변환 오류: {}", e.getMessage());
                    return new RemoteVehicleGoalResponseDto(false, "응답 메시지 형식 오류", null);
                }
            } else {
                log.error("응답 메시지 토큰 수 오류. 기대값: 3, 실제값: {}", parts.length);
                return new RemoteVehicleGoalResponseDto(false, "응답 메시지 형식 오류", null);
            }
        } else {
            log.error("응답이 없습니다.");
            return new RemoteVehicleGoalResponseDto(false, "응답이 없습니다.", null);
        }
    }

    public ParkingDestinationResponseDto getRemoteParkingDestination(ParkingDestinationRequestDto requestDto) {
        double latitude = requestDto.getDestination().getLatitude();
        double longitude = requestDto.getDestination().getLongitude();
        double radius = 5000; // 5km 반경

        List<ParkingLot> nearestList = parkingLotRepository.findTop2NearestParkingLotsWithinRadius(longitude, latitude, radius);
        List<ParkingLotDto> nearestDtos = new ArrayList<>();
        nearestList.forEach(parkingLot -> nearestDtos.add(convertToDto(parkingLot, latitude, longitude)));

        List<ParkingLot> cheapestList = parkingLotRepository.findTop2CheapestParkingLotsWithinRadius(longitude, latitude, radius);
        List<ParkingLotDto> cheapestDtos = new ArrayList<>();
        cheapestList.forEach(parkingLot -> cheapestDtos.add(convertToDto(parkingLot, latitude, longitude)));

        if (nearestDtos.isEmpty() && cheapestDtos.isEmpty()) {
            return new ParkingDestinationResponseDto(false, "주차장 리스트 조회 실패. 입력 정보를 확인해 주세요.", null, null);
        }
        return new ParkingDestinationResponseDto(true, "근처 주차장 리스트 조회 성공", nearestDtos, cheapestDtos);
    }

    private ParkingLotDto convertToDto(ParkingLot parkingLot, double destLat, double destLng) {
        double lng = parkingLot.getLocation().getX();
        double lat = parkingLot.getLocation().getY();
        double distance = calculateDistance(destLat, destLng, lat, lng);
        String congestionText = getCongestionText(parkingLot.getCongestion());
        return new ParkingLotDto(
                parkingLot.getId().toString(),
                parkingLot.getName(),
                parkingLot.getAddress(),
                lat,
                lng,
                congestionText,
                parkingLot.getBaseFee().toString(),
                parkingLot.getRateFee().toString(),
                String.format("%.1fKM", distance)
        );
    }

    private String getCongestionText(Integer congestion) {
        if (congestion == null) return "정보없음";
        if (congestion >= 80) return "혼잡";
        else if (congestion >= 50) return "보통";
        else return "여유";
    }

    private double calculateDistance(double lat1, double lon1, double lat2, double lon2) {
        final int R = 6371;
        double latDistance = Math.toRadians(lat2 - lat1);
        double lonDistance = Math.toRadians(lon2 - lon1);
        double a = Math.sin(latDistance / 2) * Math.sin(latDistance / 2)
                + Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2))
                * Math.sin(lonDistance / 2) * Math.sin(lonDistance / 2);
        double c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
        return R * c;
    }

    // 배회 요청 처리 (토큰은 저장하지 않고, 매개변수로 전달)
    public RemoteVehicleRoamResponseDto processRoamRequest(RemoteVehicleRoamRequestDto requestDto, Long userNo, String token) {
        Instant startTime = Instant.now();
        Instant endTime = startTime.plusSeconds(requestDto.getRoamDuration() * 60);
        String message = "Roam";
        rabbitTemplate.convertAndSend(remoteExchange, roamRoutingKey, message);

        // RoamTask에는 토큰을 저장하지 않음
        Optional<RoamTask> existingTaskOpt = roamTaskRepository.findByVehicleVin(requestDto.getVehicleVin());

        RoamTask task = new RoamTask();
        task.setVehicleVin(requestDto.getVehicleVin());
        task.setUserNo(userNo);
        task.setRoamStartTime(startTime);
        task.setRoamEndTime(endTime);
        task.setBoarded(false);
        task.setUserResponded(false);

        if (existingTaskOpt.isPresent()) {
            // 기존 엔티티가 있다면 필요한 필드를 갱신
            RoamTask existingTask = existingTaskOpt.get();
            existingTask.setExtendedRoamEndTime(task.getExtendedRoamEndTime());
            existingTask.setRoamEndTime(task.getRoamEndTime());
            existingTask.setRoamStartTime(task.getRoamStartTime());
            existingTask.setUserNo(task.getUserNo());
            // 업데이트 처리
            roamTaskRepository.save(existingTask);
        }else {
            roamTaskRepository.save(task);
        }

        // 종료시각에 상태 체크 예약, token 전달하고 ScheduledFuture 저장
        ScheduledFuture<?> future = taskScheduler.schedule(() -> checkRoamStatusAndNotify(task.getId(), token, requestDto.getVehicleVin()), Date.from(endTime));
        scheduledTasks.computeIfAbsent(requestDto.getVehicleVin(), k -> new ArrayList<>())
                .add(future);

        return new RemoteVehicleRoamResponseDto(true, "배회 시작됨. 지정된 시간 동안 차량이 배회합니다.");
    }

    // 배회 종료 후 첫 번째 알림 처리 (토큰을 매개변수로 전달)
    private void checkRoamStatusAndNotify(Long taskId, String token, String vehicleVin) {
        Optional<RoamTask> opt = roamTaskRepository.findById(taskId);
        if (opt.isEmpty()) return;
        RoamTask task = opt.get();
        if (task.isBoarded()) return;

        // 버튼 액션 정보를 구성합니다.
        Map<String, String> actions = Map.of(
                "yes", "KEEP_ROAM",
                "no", "PARKING_API"
        );
        NotificationRequestDto notifDto = new NotificationRequestDto(
                "[배회 중 알림]",
                "배회 요청 시간이 지났습니다.",
                actions
        );
        notificationServiceClient.sendPushNotification(token, notifDto);

        long originalDuration = task.getRoamEndTime().getEpochSecond() - task.getRoamStartTime().getEpochSecond();
        Instant secondCheckTime = task.getRoamEndTime().plusSeconds(originalDuration);

        // 두 번째 알림 예약 및 ScheduledFuture 저장
        ScheduledFuture<?> secondFuture = taskScheduler.schedule(() -> checkUserResponseAndPark(task.getId(), token, vehicleVin), Date.from(secondCheckTime));
        scheduledTasks.computeIfAbsent(task.getVehicleVin(), k -> new ArrayList<>()).add(secondFuture);
    }

    // 두 번째 알림 처리 (토큰 전달)
    private void checkUserResponseAndPark(Long taskId, String token, String vehicleVin) {
        Optional<RoamTask> opt = roamTaskRepository.findById(taskId);
        if (opt.isEmpty()) return;
        RoamTask task = opt.get();
        if (task.isUserResponded()) return;

        NotificationRequestDto noti = new NotificationRequestDto(
                "[자동 주차 안내]",
                "지정 시간이 지나 자동으로 주차장으로 이동합니다.",
                null
        );
        stopRoamService(vehicleVin, token);
        notificationServiceClient.sendPushNotification(token, noti);

        List<ParkingLot> lots = parkingLotRepository.findTop2NearestParkingLotsWithinRadius(126.5628378, 37.5249278, 5000);
        if (!lots.isEmpty()) {
            ParkingLot parkingLot = lots.get(0);
            double latitude = parkingLot.getLocation().getY();
            double longitude = parkingLot.getLocation().getX();
            sendParkingInstruction(token, latitude, longitude);
        }

        // 두 번째 알림 후 RoamTask 삭제
        roamTaskRepository.delete(task);
    }


    // 배회 연장 요청 시, 토큰을 함께 전달받아 스케줄러 콜백에 사용
    public void handleUserResponse(String vehicleVin, long additionalTime, String token) {
        Optional<RoamTask> optionalTask = roamTaskRepository.findByVehicleVin(vehicleVin);
        if (optionalTask.isEmpty()) {
            return;
        }
        RoamTask task = optionalTask.get();
        task.setUserResponded(true);

        Instant newRoamEndTime = task.getRoamEndTime().plusSeconds(additionalTime * 60);
        task.setExtendedRoamEndTime(newRoamEndTime);

        // RoamTask에는 토큰을 저장하지 않음
        Optional<RoamTask> existingTaskOpt = roamTaskRepository.findByVehicleVin(task.getVehicleVin());

        if (existingTaskOpt.isPresent()) {
            // 기존 엔티티가 있다면 필요한 필드를 갱신

            existingTask.setRoamEndTime(task.getRoamEndTime());
            existingTask.setRoamStartTime(task.getRoamStartTime());
            existingTask.setUserNo(task.getUserNo());
            // 업데이트 처리
            roamTaskRepository.save(existingTask);
        }else {
            roamTaskRepository.save(task);
        }

        ScheduledFuture<?> additionalFuture = taskScheduler.schedule(() -> checkRoamStatusAndNotify(task.getId(), token, vehicleVin), Date.from(newRoamEndTime));
        scheduledTasks.computeIfAbsent(vehicleVin, k -> new ArrayList<>()).add(additionalFuture);
    }

    public void stopRoamService(@NotBlank(message = "차량 VIN은 필수 입력 값입니다.") String vehicleVin, String authorization) {
        String message = "Stop";
        rabbitTemplate.convertAndSend(remoteExchange, roamRoutingKey, message);

        // 예약된 스케줄 작업 취소
        List<ScheduledFuture<?>> futures = scheduledTasks.get(vehicleVin);
        if (futures != null) {
            for (ScheduledFuture<?> future : futures) {
                future.cancel(true);
            }
            scheduledTasks.remove(vehicleVin);
        }

        // 해당 RoamTask 삭제
        Optional<RoamTask> optionalTask = roamTaskRepository.findByVehicleVin(vehicleVin);
        optionalTask.ifPresent(task -> roamTaskRepository.delete(task));
    }

    /**
     * 배회 중지 및 차량 위치와 사용자 위치를 기반으로 가장 가까운 주차장을 찾아
     * 주차 명령 메시지를 전송한 후, 해당 주차장의 정보를 리스폰스로 반환하는 메서드.
     *
     * @param vehicleVin 차량 VIN
     * @param authorization JWT 토큰 (또는 인증 문자열)
     * @param vehicleLat 차량의 현재 위도
     * @param vehicleLon 차량의 현재 경도
     * @param userLat 사용자의 현재 위도
     * @param userLon 사용자의 현재 경도
     * @return ParkingLotResponseDto - 주차장 이름, 주소, 위치, 혼잡도, 기본/추가 요금 및
     *         사용자의 위치와 주차장 사이의 거리 정보를 포함한 DTO.
     */
    public ParkingLotResponseDto stopRoamAndPark(@NotBlank(message = "차량 VIN은 필수 입력 값입니다.") String vehicleVin,
                                                 String authorization,
                                                 double vehicleLat,
                                                 double vehicleLon,
                                                 double userLat,
                                                 double userLon) {
        // 1. 배회 중지 처리
        stopRoamService(vehicleVin, authorization);

        // 2. 차량 위치를 기준으로 주차장 검색 (5km 반경)
        double radius = 5000;
        List<ParkingLot> parkingLots = parkingLotRepository.findTop2NearestParkingLotsWithinRadius(vehicleLon, vehicleLat, radius);
        if (parkingLots.isEmpty()) {
            log.warn("No parking lot found within radius {} for vehicle location (lat={}, lon={})", radius, vehicleLat, vehicleLon);
            return null;
        }
        ParkingLot nearest = parkingLots.get(0);
        // 주차장의 위치 정보 (ParkingLot 엔티티에서는 위도는 Y, 경도는 X)
        double parkingLat = nearest.getLocation().getY();
        double parkingLon = nearest.getLocation().getX();

        // 3. 주차 명령 메시지를 전송
        sendParkingInstruction(authorization, parkingLat, parkingLon);
        log.info("Nearest parking lot found. Sending parking instruction to lat={}, lon={}", parkingLat, parkingLon);

        // 4. 사용자 위치와 주차장 위치 사이의 거리를 계산 (calculateDistance 메서드 활용)
        double distanceValue = calculateDistance(userLat, userLon, parkingLat, parkingLon);
        String distanceStr = String.format("%.1fKM", distanceValue);

        // 5. 조회된 주차장의 정보와 계산된 거리를 포함하여 ParkingLotResponseDto 생성
        ParkingLotResponseDto responseDto = new ParkingLotResponseDto(
                nearest.getName(),                             // parkingLotName
                nearest.getAddress(),                          // address
                parkingLat,                                    // latitude
                parkingLon,                                    // longitude
                getCongestionText(nearest.getCongestion()),    // congestion
                nearest.getBaseFee().toString(),               // baseFee
                nearest.getRateFee().toString(),               // rateFee
                distanceStr                                    // distance (사용자 위치와 주차장 사이)
        );
        return responseDto;
    }

package com.moonshot.remote_service.util;

import com.moonshot.remote_service.exception.JwtValidationException;
import io.jsonwebtoken.Claims;
import io.jsonwebtoken.ExpiredJwtException;
import io.jsonwebtoken.Jwts;
import jakarta.annotation.PostConstruct;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.io.ClassPathResource;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;

import java.security.KeyFactory;
import java.security.PublicKey;
import java.security.spec.X509EncodedKeySpec;
import java.time.Instant;
import java.util.Base64;
import java.util.Date;

@Slf4j
@Component
public class JWTUtil {

    @Value("${jwt.public-key-path}")
    private String publicKeyPath;
    private PublicKey publicKey;

    @PostConstruct // Bean이 생성되고, Dependency Injection이 완료된 후 초기화 해야함을 명시.
    public void init() throws Exception {
        this.publicKey = loadPublicKey();
    }

    private PublicKey loadPublicKey() throws Exception {
        ClassPathResource resource = new ClassPathResource(publicKeyPath);

        String publicKeyPEM = new String(resource.getInputStream().readAllBytes())
                .replace("-----BEGIN PUBLIC KEY-----", "")
                .replace("-----END PUBLIC KEY-----", "")
                .replaceAll("\\s", "");

        byte[] encoded = Base64.getDecoder().decode(publicKeyPEM);
        KeyFactory keyFactory = KeyFactory.getInstance("RSA");
        X509EncodedKeySpec keySpec = new X509EncodedKeySpec(encoded);

        return keyFactory.generatePublic(keySpec);
    }

    public Long extractUserNo(String token, boolean isRefreshToken) {
        return Long.parseLong(extractAllClaims(token, isRefreshToken).getSubject());
    }

    public Claims extractAllClaims(String token, boolean isRefreshToken) {
        String actualToken = null;
        try {
            log.info("token: '{}'", token);
            log.info("isRefreshToken= '{}'", isRefreshToken);
            // refreshToken인 경우 그대로 사용, accessToken인 경우 Bearer 제거
            actualToken = isRefreshToken ? token : resolveToken(token);
            log.info("token: '{}'", actualToken);
            return Jwts.parser()
                    .clock(() -> Date.from(Instant.now()))
                    .verifyWith(publicKey)
                    .build()
                    .parseSignedClaims(actualToken)
                    .getPayload();

        } catch (ExpiredJwtException e) {
            throw new JwtValidationException(
                    JwtValidationException.JwtErrorType.EXPIRED_TOKEN,
                    e
            );
        }
    }

    // Bearer 토큰에서 JWT 추출
    public String resolveToken(String bearerToken) {
        log.info("bearerToken: '{}'", bearerToken);

        // Bearer 접두사가 없는 경우도 처리
        if (StringUtils.hasText(bearerToken)) {
            if (bearerToken.startsWith("Bearer ")) {
                return bearerToken.substring(7).trim();
            } else {
                // Bearer 접두사 없이 토큰이 직접 전달된 경우
                return bearerToken.trim();
            }
        }
        return null;
    }
}

     * 차량 등록 API
     *
     * 클라이언트는 Authorization 헤더에 담긴 JWT에서 회원 ID를 추출하고,
     * 차량 등록 정보를 전송한다.
     * 검증 및 등록에 성공하면 "등록 성공!" 메시지를, 오류 발생 시 "등록 실패!" 메시지를 반환한다.
     *
     * @param authorization "Bearer {accessToken}" 형식의 헤더
     * @param request       차량 등록 요청 DTO
     * @return 성공 시 "등록 성공!", 실패 시 "등록 실패!" 메시지
     */
    @PostMapping("/register")
    public ResponseEntity<String> registerVehicle(
            @RequestHeader("Authorization") String authorization,
            @Valid @RequestBody VehicleRegisterRequest request) {
        try {
            String token = jwtUtil.resolveToken(authorization);
            if (token == null || token.isEmpty()) {
                return ResponseEntity.badRequest().body("등록 실패! (유효한 토큰이 없습니다.)");
            }
            // JWT에서 회원 ID 추출 (Access Token 처리, isRefreshToken=false)
            Long ownerId = jwtUtil.extractUserNo(token, false);
            log.info("ownerId: {}", ownerId);
            vehicleService.registerVehicle(ownerId, request);
            return ResponseEntity.ok("등록 성공!");
        } catch (IllegalArgumentException e) {
            log.error("차량 등록 오류: {}", e.getMessage());
            return ResponseEntity.badRequest().body("등록 실패! " + e.getMessage());
        } catch (Exception e) {
            log.error("차량 등록 중 예외 발생", e);
            return ResponseEntity.internalServerError().body("등록 실패! 예외 발생");

     *
     * @param vehicleVin 차량 차대번호
     * @return VehicleStatusResponse 객체
     */
    public VehicleStatusResponse getVehicleStatus(String vehicleVin, VehicleStatusRequest request) {
        String key = "vehicle:" + vehicleVin + ":status";
        HashOperations<String, String, Object> hashOps = redisTemplate.opsForHash();
        Object parkedObj = hashOps.get(key, "isParked");
        Object engineStatusObj = hashOps.get(key, "engineStatus");
        Object tsObj = hashOps.get(key, "timestamp");
        Object speedKmhObj = hashOps.get(key, "speedKmh");

        if (parkedObj == null || engineStatusObj == null || tsObj == null || speedKmhObj == null) {
            throw new IllegalArgumentException("해당 차량의 상태 정보가 존재하지 않습니다.");
        }

        boolean isParked = Boolean.parseBoolean(parkedObj.toString());
        String engineStatus = engineStatusObj.toString();
        String timestamp = tsObj.toString();
        double speedKmh = Double.parseDouble(speedKmhObj.toString());

        // 차량 GPS (실제 차량 위치, carGps)
        double vehicleLat = request.getCarGps().getLatitude();
        double vehicleLon = request.getCarGps().getLongitude();

        // 클라이언트 GPS (일반 gps, 사용자의 위치)
        double clientLat = request.getGps().getLatitude();
        double clientLon = request.getGps().getLongitude();

        // 주차장 위치 (고정 좌표)
        double lotLat = 37.5665;
        double lotLon = 126.9780;

        double kmDistanceToCar = calculateDistance(vehicleLat, vehicleLon, clientLat, clientLon);
        double kmDistanceToLot = calculateDistance(vehicleLat, vehicleLon, lotLat, lotLon);

        // km를 미터로 변환
        double distanceToCar = kmDistanceToCar * 1000;
        double distanceToLot = kmDistanceToLot * 1000;

        return new VehicleStatusResponse(vehicleVin, isParked, engineStatus, timestamp, distanceToLot, distanceToCar, speedKmh);

    /**
     * 두 지점 간의 거리를 Haversine 공식을 이용해 계산.
     * 거리는 km 단위로 반환.
     *
     * @param lat1 첫 번째 지점의 위도
     * @param lon1 첫 번째 지점의 경도
     * @param lat2 두 번째 지점의 위도
     * @param lon2 두 번째 지점의 경도
     * @return 두 지점 간의 거리 (km)
     */
    private double calculateDistance(double lat1, double lon1, double lat2, double lon2) {
        final int R = 6371; // 지구의 반지름 (km)
        double latDistance = Math.toRadians(lat2 - lat1);
        double lonDistance = Math.toRadians(lon2 - lon1);
        double a = Math.sin(latDistance / 2) * Math.sin(latDistance / 2)
                + Math.cos(Math.toRadians(lat1)) * Math.cos(Math.toRadians(lat2))
                * Math.sin(lonDistance / 2) * Math.sin(lonDistance / 2);
        double c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
        return R * c;
    }

package com.moonshot.vehicle_service.util;

import com.moonshot.vehicle_service.exception.JwtValidationException;
import io.jsonwebtoken.Claims;
import io.jsonwebtoken.ExpiredJwtException;
import io.jsonwebtoken.Jwts;
import jakarta.annotation.PostConstruct;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.core.io.ClassPathResource;
import org.springframework.stereotype.Component;
import org.springframework.util.StringUtils;

import java.security.KeyFactory;
import java.security.PublicKey;
import java.security.spec.X509EncodedKeySpec;
import java.time.Instant;
import java.util.Base64;
import java.util.Date;

@Slf4j
@Component
public class JWTUtil {

    @Value("${jwt.public-key-path}")
    private String publicKeyPath;
    private PublicKey publicKey;

    @PostConstruct // Bean이 생성되고, Dependency Injection이 완료된 후 초기화 해야함을 명시.
    public void init() throws Exception {
        this.publicKey = loadPublicKey();
    }

    private PublicKey loadPublicKey() throws Exception {
        ClassPathResource resource = new ClassPathResource(publicKeyPath);

        String publicKeyPEM = new String(resource.getInputStream().readAllBytes())
                .replace("-----BEGIN PUBLIC KEY-----", "")
                .replace("-----END PUBLIC KEY-----", "")
                .replaceAll("\\s", "");

        byte[] encoded = Base64.getDecoder().decode(publicKeyPEM);
        KeyFactory keyFactory = KeyFactory.getInstance("RSA");
        X509EncodedKeySpec keySpec = new X509EncodedKeySpec(encoded);

        return keyFactory.generatePublic(keySpec);
    }

    public Long extractUserNo(String token, boolean isRefreshToken) {
        return Long.parseLong(extractAllClaims(token, isRefreshToken).getSubject());
    }

    public Claims extractAllClaims(String token, boolean isRefreshToken) {
        String actualToken = null;
        try {
            log.info("token: '{}'", token);
            log.info("isRefreshToken= '{}'", isRefreshToken);
            // refreshToken인 경우 그대로 사용, accessToken인 경우 Bearer 제거
            actualToken = isRefreshToken ? token : resolveToken(token);
            log.info("token: '{}'", actualToken);
            return Jwts.parser()
                    .clock(() -> Date.from(Instant.now()))
                    .verifyWith(publicKey)
                    .build()
                    .parseSignedClaims(actualToken)
                    .getPayload();

        } catch (ExpiredJwtException e) {
            throw new JwtValidationException(
                    JwtValidationException.JwtErrorType.EXPIRED_TOKEN,
                    e
            );
        }
    }

    // Bearer 토큰에서 JWT 추출
    public String resolveToken(String bearerToken) {
        log.info("bearerToken: '{}'", bearerToken);

        // Bearer 접두사가 없는 경우도 처리
        if (StringUtils.hasText(bearerToken)) {
            if (bearerToken.startsWith("Bearer ")) {
                return bearerToken.substring(7).trim();
            } else {
                // Bearer 접두사 없이 토큰이 직접 전달된 경우
                return bearerToken.trim();
            }
        }
        return null;
    }
}
