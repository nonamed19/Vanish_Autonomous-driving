#!/usr/bin/env python
# -*- coding: utf-8 -*-

import rospy
import math
from math import cos, sin, sqrt
from geometry_msgs.msg import Point
from nav_msgs.msg import Odometry, Path
from morai_msgs.msg import CtrlCmd, EgoVehicleStatus, ObjectStatusList
import numpy as np
from tf.transformations import euler_from_quaternion
from std_msgs.msg import Int32,String
from morai_msgs.srv import MoraiEventCmdSrv
from morai_msgs.msg import EventInfo

from pid_control import pidControl
from velocityPlanning import velocityPlanning
from AdaptiveCruiseControl import AdaptiveCruiseControl
import time


class pure_pursuit:
    def __init__(self):
        rospy.init_node('pure_pursuit', anonymous=True)

        # (1) Subscriber
        rospy.Subscriber("/global_path", Path, self.global_path_callback)
        rospy.Subscriber("/local_path", Path, self.path_callback)
        rospy.Subscriber("/odom", Odometry, self.odom_callback)
        rospy.Subscriber("/Ego_topic", EgoVehicleStatus, self.status_callback)
        rospy.Subscriber("/Object_topic", ObjectStatusList, self.object_info_callback)
        rospy.Subscriber("/arrival_status", Int32, self.arrival_status_callback)
        
        # **추가**: 신호등 decision(0 or 1) 구독
        rospy.Subscriber("/go_decision", Int32, self.decision_callback)
        # **추가**: way(-1=좌회전,0=직진,1=우회전) 구독 (필요 시)
        rospy.Subscriber("/way", Int32, self.way_callback)

        rospy.wait_for_service('/Service_MoraiEventCmd')
        self.event_cmd_srv = rospy.ServiceProxy('/Service_MoraiEventCmd', MoraiEventCmdSrv)
        
        # (2) Publisher
        self.ctrl_cmd_pub = rospy.Publisher("/ctrl_cmd", CtrlCmd, queue_size=10)
        
        rospy.Subscriber("/parking_trigger", String, self.trigger_callback)
        self.is_parking = True

        # (3) CtrlCmd 메시지
        self.ctrl_cmd_msg = CtrlCmd()
        self.ctrl_cmd_msg.longlCmdType = 1  # Throttle/Brake 제어

        # (4) 상태 플래그
        self.is_path = False
        self.is_odom = False
        self.is_status = False
        self.is_global_path = False
        self.is_look_forward_point = False
        self.is_object_info = False
        self.is_arrive = 0

        # (5) 차량 위치/자세
        self.forward_point = Point()
        self.current_postion = Point()
        self.vehicle_yaw = 0.0

        # (6) Pure Pursuit 파라미터
        self.vehicle_length = 2.6
        self.lfd = 8
        self.min_lfd = 5
        self.max_lfd = 30
        self.lfd_gain = 0.85
        self.target_velocity = 45  # km/h

        # (7) 객체 데이터
        self.object_data = None

        # (8) 종방향 PID, 곡률 기반 속도계획
        self.speed_pid = pidControl(p_gain=0.25, i_gain=0.0, d_gain=0.03)   # 속도용
        self.steering_pid = pidControl(p_gain=0.4, i_gain=0.0, d_gain=0.0025)  # 조향용
        self.vel_planning = velocityPlanning(self.target_velocity / 3.6, 0.15)
        self.adaptive_cruise_control = AdaptiveCruiseControl(velocity_gain=0.5,
                                                             distance_gain=1,
                                                             time_gap=0.8,
                                                             vehicle_length=2.7)

        # (9) 신호등 decision(0/1) 기본값: 1(주행 가능)
        self.decision = 1
        self.way = 0  # -1=좌회전, 0=직진, 1=우회전 (필요하다면 사용)

        # (10) 전역 경로 수신 대기 후 속도 계획
        while not rospy.is_shutdown():
            if self.is_global_path:
                break
            else:
                rospy.loginfo('Waiting for global path data...')
                rospy.sleep(0.5)

        # (11) 메인 루프
        rate = rospy.Rate(30)
        while not rospy.is_shutdown():
            if self.is_path and self.is_odom and self.is_status and self.is_parking  :
                
                # (A) 도착 여부 체크
                if self.is_arrive == 2:
                    rospy.loginfo("목적지에 도착 => 차량 정지")
                    self.ctrl_cmd_msg.steering = 0.0
                    self.ctrl_cmd_msg.accel = 0.0
                    self.ctrl_cmd_msg.brake = 1.0
                    self.ctrl_cmd_pub.publish(self.ctrl_cmd_msg)
                    self.get_gear_status(1)
                    self.is_global_path = False
                    rate.sleep()
                    continue
                
                # (B) 주행 로직
                if self.is_global_path:
                    self.get_gear_status(4)
                    result = self.calc_vaild_obj([self.current_postion.x,self.current_postion.y,self.vehicle_yaw],self.object_data)
                
                    global_npc_info = result[0] 
                    local_npc_info = result[1] 
                    global_ped_info = result[2] 
                    local_ped_info = result[3] 
                    global_obs_info = result[4] 
                    local_obs_info = result[5] 
                    
                    # (1) 현재 waypoint 찾기
                    self.current_waypoint = self.get_current_waypoint(
                        [self.current_postion.x, self.current_postion.y],
                        self.global_path
                    )
                    idx = min(self.current_waypoint, len(self.velocity_list) - 1)
                    
                    # (2) 원래 속도 계획
                    self.target_velocity = self.velocity_list[idx] * 3.6
                    
                    # (3) 신호등 decision 확인
                    #     0 => 최종 속도 = 0 (서서히 감속)
                    #     1 => 기존 계획 속도 유지
                    if self.decision == 0:
                        self.target_velocity = 0.0
                        
                        
                    self.adaptive_cruise_control.check_object(self.path ,global_npc_info, local_npc_info,global_ped_info, local_ped_info,global_obs_info, local_obs_info)
                    
                    self.target_velocity = self.adaptive_cruise_control.get_target_velocity(local_npc_info, local_ped_info, local_obs_info,
                                                                                                        self.status_msg.velocity.x, self.target_velocity/3.6)


                    # (4) 조향(Pure Pursuit)
                    steering_pp = self.calc_pure_pursuit()
                    wheel_angle_deg = self.status_msg.wheel_angle
                    wheel_angle_rad = wheel_angle_deg * (math.pi / 180.0)
                    steering = self.steering_pid.pid(steering_pp, wheel_angle_rad)
                    
                    if self.is_look_forward_point:
                        self.ctrl_cmd_msg.steering = steering
                    else:
                        rospy.loginfo("no forward point => steering=0")
                        #self.ctrl_cmd_msg.steering = 0.0

                    # (5) 종방향 PID
                    current_speed = self.status_msg.velocity.x * 3.6  # m/s -> km/h
                    output = self.speed_pid.pid(self.target_velocity, current_speed)

                    if output > 0.0:
                        self.ctrl_cmd_msg.accel = output
                        self.ctrl_cmd_msg.brake = 0.0
                    else:

    def calc_pure_pursuit(self):
        # (1) 속도 기반 LfD 조정
        vehicle_speed = abs(self.status_msg.velocity.x)  # m/s
        self.lfd = max(self.min_lfd, min(vehicle_speed * self.lfd_gain, self.max_lfd))

        # (2) 차량 좌표계 변환
        self.is_look_forward_point = False
        vehicle_position = self.current_postion
        trans_matrix = np.array([
            [math.cos(self.vehicle_yaw), -math.sin(self.vehicle_yaw), vehicle_position.x],
            [math.sin(self.vehicle_yaw),  math.cos(self.vehicle_yaw), vehicle_position.y],
            [0, 0, 1]
        ])
        det_trans_matrix = np.linalg.inv(trans_matrix)

        # (3) local 좌표로 변환하며 LfD 이상 떨어진 점 찾기
        for pose_stamped in self.path.poses:
            path_point = pose_stamped.pose.position
            global_point = np.array([path_point.x, path_point.y, 1.0])
            local_point = det_trans_matrix.dot(global_point)
            if local_point[0] > 0:
                dist = math.sqrt(local_point[0]**2 + local_point[1]**2)
                if dist >= self.lfd:
                    self.forward_point = path_point
                    self.is_look_forward_point = True
                    break

        # (4) 전방 주시점 없으면 조향 0
        if not self.is_look_forward_point:
            return 0.0

        # (5) 조향각 계산
        theta = math.atan2(self.forward_point.y - vehicle_position.y,
                           self.forward_point.x - vehicle_position.x)
        alpha = theta - self.vehicle_yaw
        steering = math.atan2(2.0 * self.vehicle_length * math.sin(alpha), self.lfd)
        return steering

# -*- coding: utf-8 -*-

import rospy

from nav_msgs.msg import Path, Odometry

from std_msgs.msg import Int32
from tf.transformations import euler_from_quaternion


class ArrivalDetector:
    def __init__(self):
        rospy.init_node('arrival_detector')
        
        # 토픽 구독 및 퍼블리셔 생성
        rospy.Subscriber("/global_path", Path, self.path_callback)
        rospy.Subscriber("/odom", Odometry, self.odom_callback)
        self.state_pub = rospy.Publisher("/arrival_status", Int32, queue_size=10)
        
        # 변수 초기화
        self.global_path = []       # PoseStamped 리스트
        self.current_pose = Pose()  # 현재 차량 위치 (geometry_msgs/Pose)
        self.vehicle_yaw = 0.0
        self.is_global_path = False
        
        # 새로운 경로가 들어오면 arrived 상태를 해제하기 위해 사용하는 변수
        self.arrived = False
        
        # (출발/도착 여부를 판별하기 위한 파라미터들)
        self.min_remaining_points = 5  # 남은 경로 포인트 임계치 (이하이면 도착 간주)

    def path_callback(self, msg):
        """
        글로벌 경로 콜백
          - 새 경로가 들어오면 self.arrived = False로 초기화
          - 이후 get_state()에서 다시 출발(0) 또는 주행(1) 상태로 변경 가능
        """
        self.global_path = msg.poses
        self.is_global_path = len(self.global_path) > 0
        
        if self.is_global_path:
            # 새로운 경로가 들어왔으니 도착 상태 해제
            self.arrived = False
            rospy.loginfo("새로운 글로벌 경로 수신됨 => 도착 상태 해제, 출발(주행) 가능 상태로 복귀")
        else:
            rospy.loginfo("수신된 경로가 없습니다.")

    def odom_callback(self, msg):
        # odom 토픽에서 현재 위치 및 orientation 업데이트
        orientation_q = msg.pose.pose.orientation
        quaternion = (orientation_q.x, orientation_q.y, orientation_q.z, orientation_q.w)
        _, _, self.vehicle_yaw = euler_from_quaternion(quaternion)
        
        self.current_pose.position.x = msg.pose.pose.position.x
        self.current_pose.position.y = msg.pose.pose.position.y
        
        # 현재 상태 계산 후 퍼블리시
        state = self.get_state()
        self.publish_state(state)

    def get_state(self):
        """
        상태 판단 (0=출발, 1=주행중, 2=도착)
        
        1) 만약 글로벌 경로가 없다면 => 0(출발)로 간주
        2) 이미 도착 상태라면 => 2(도착) 유지
        3) 도착 판정이 되면 => arrived=True로 바꾸고 2(도착) 반환
        4) 아직 도착 전이면, 현재 waypoint 인덱스가 10 이하 => 0(출발), 그 외 => 1(주행)
        """
        if not self.is_global_path or len(self.global_path) == 0:
            # 경로가 없으면 '출발' 상태로 간주 (혹은 상황에 따라 1(주행중)으로 보고 싶다면 조정 가능)
            return 0
        
        # 이미 도착한 상태(arrived=True)면 계속 2(도착) 유지
        if self.arrived:
            return 2

        # 아직 arrived=False인데, 실제 도착 판정이 났는지 확인
        if self.is_arrived(threshold=1.0) and self.check_remaining_points():
            self.arrived = True
            return 2
        
        # 도착도 아니고, 경로가 존재한다면 출발 or 주행 상태 구분
        ego_xy = [self.current_pose.position.x, self.current_pose.position.y]
        current_idx = self.get_current_waypoint(ego_xy, self.global_path)

        if current_idx <= 10:
            return 0  # 출발 상태
        else:
            return 1  # 주행중

    def is_arrived(self, threshold=1.0):
        """
        목적지(글로벌 경로의 마지막 포인트)와 현재 위치가 threshold 미만이면 도착으로 판정
        """
        if not self.is_global_path or len(self.global_path) == 0:
            return False
        goal_pos = self.global_path[-1].pose.position
        dx = goal_pos.x - self.current_pose.position.x
        dy = goal_pos.y - self.current_pose.position.y
        dist_to_goal = math.sqrt(dx**2 + dy**2)
        return dist_to_goal < threshold

    def check_remaining_points(self):
        """
        남은 글로벌 경로 포인트 수가 min_remaining_points 미만이면 도착으로 판정
        """
        if not self.is_global_path or len(self.global_path) == 0:
            return False
        
        ego_xy = [self.current_pose.position.x, self.current_pose.position.y]
        current_idx = self.get_current_waypoint(ego_xy, self.global_path)
        remaining_points = len(self.global_path) - current_idx
        return remaining_points < self.min_remaining_points

    def get_current_waypoint(self, ego_xy, global_path):
        """
        ego_xy : [x, y]
        global_path : PoseStamped 리스트
        => ego_xy와 가장 가까운 waypoint 인덱스 반환
        """
        min_dist = float('inf')
        current_waypoint = -1
        for i, pose in enumerate(global_path):
            dx = ego_xy[0] - pose.pose.position.x
            dy = ego_xy[1] - pose.pose.position.y
            dist = math.sqrt(dx*dx + dy*dy)
            if dist < min_dist:
                min_dist = dist
                current_waypoint = i
        return current_waypoint

    def publish_state(self, state):
        msg = Int32()
        msg.data = state
        self.state_pub.publish(msg)
        state_str = {0: "출발", 1: "주행중", 2: "도착"}.get(state, "알수없음")
        rospy.loginfo("상태: %s (%d)", state_str, state)

import heapq
from math import sqrt, pow

class AStar:
    """
    Dijkstra -> A* 로 수정한 클래스.
    1) weight_matrix로 from_node->to_node 이동비용을 계산
    2) find_shortest_path()에서 A* 알고리즘 적용
    """
    def __init__(self, nodes, links):
        self.nodes = nodes
        self.links = links
        # (1) from_node->to_node 비용 계산 (inf 아닌 곳만 "이웃")
        self.weight = self.get_weight_matrix()

    def get_weight_matrix(self):
        """
        노드 간 이동비용(거리)을 담은 dict.
        weight[a][b] = (a->b 이동코스트 or inf)
        """
        weight = {
            from_node_id: {to_node_id: float('inf') for to_node_id in self.nodes.keys()}
            for from_node_id in self.nodes.keys()
        }

        for from_node_id, from_node in self.nodes.items():
            # 자기 자신->자신 비용=0
            weight[from_node_id][from_node_id] = 0.0

            # from_node에서 갈 수 있는 모든 to_node
            for to_node in from_node.get_to_nodes():
                shortest_link, min_cost = self.find_shortest_link_leading_to_node(from_node, to_node)
                weight[from_node_id][to_node.idx] = min_cost

        return weight

    def find_shortest_link_leading_to_node(self, from_node, to_node):
        """ from_node -> to_node 중 가장 짧은 link를 찾음 """
        min_cost = float('inf')
        shortest_link = None

        for link in from_node.get_to_links():
            if link.to_node.idx == to_node.idx:
                cost = 0.0
                for i in range(len(link.points) - 1):
                    x1, y1 = link.points[i][0], link.points[i][1]
                    x2, y2 = link.points[i+1][0], link.points[i+1][1]
                    cost += sqrt(pow(x2 - x1, 2) + pow(y2 - y1, 2))

                if cost < min_cost:
                    min_cost = cost
                    shortest_link = link

        return shortest_link, min_cost

    def find_shortest_path(self, start_node_idx, end_node_idx):
        """
        A* 알고리즘으로 start_node ~ end_node 최단 경로 찾기
        return:
          result(True/False), { 'node_path':..., 'link_path':..., 'point_path':... }
        """
        # 특수 케이스: 시작==목표
        if start_node_idx == end_node_idx:
            return True, {
                'node_path': [start_node_idx],
                'link_path': [],
                'point_path': [
                    [self.nodes[start_node_idx].point[0],
                     self.nodes[start_node_idx].point[1],
                     0.0]
                ]
            }

        # open/closed set
        open_set = set([start_node_idx])
        closed_set = set()

        # g(n): 시작~현재 노드까지 실제 비용
        g_score = {n: float('inf') for n in self.nodes.keys()}
        g_score[start_node_idx] = 0.0

        # f(n)=g(n)+h(n): 현재까지 비용 + 휴리스틱
        f_score = {n: float('inf') for n in self.nodes.keys()}
        f_score[start_node_idx] = self.heuristic(self.nodes[start_node_idx], self.nodes[end_node_idx])

        # 경로 추적용
        came_from = {}  # { node_id: previous_node_id }

        while open_set:
            # f_score가 가장 작은 노드를 꺼냄
            current = min(open_set, key=lambda x: f_score[x])

            # 목표 도달?
            if current == end_node_idx:
                return True, self.reconstruct_path(came_from, current, start_node_idx, end_node_idx)

            open_set.remove(current)
            closed_set.add(current)

            # 이웃 노드 순회
            for neighbor in self.nodes.keys():
                # 비용이 inf가 아니면 neighbor로 갈 수 있음
                cost = self.weight[current][neighbor]
                if cost == float('inf'):
                    continue
                if neighbor in closed_set:
                    continue

                tentative_g = g_score[current] + cost
                if tentative_g < g_score[neighbor]:
                    came_from[neighbor] = current
                    g_score[neighbor] = tentative_g
                    # f(n) = g(n) + h(n)
                    f_score[neighbor] = tentative_g + self.heuristic(self.nodes[neighbor], self.nodes[end_node_idx])

                    if neighbor not in open_set:
                        open_set.add(neighbor)

        # 실패
        return False, {}

    def reconstruct_path(self, came_from, current, start_node_idx, end_node_idx):
        """
        A* 완료 후, 'came_from' 정보로 node_path / link_path / point_path 구성
        """
        node_path = [current]
        while current in came_from:
            current = came_from[current]
            node_path.append(current)
        node_path.reverse()

        # link_path: 연속된 node 쌍에 대해 find_shortest_link_leading_to_node
        link_path = []
        for i in range(len(node_path) - 1):
            f_node_id = node_path[i]
            t_node_id = node_path[i+1]
            f_node_obj = self.nodes[f_node_id]
            t_node_obj = self.nodes[t_node_id]

            shortest_link, _ = self.find_shortest_link_leading_to_node(f_node_obj, t_node_obj)
            if shortest_link:
                link_path.append(shortest_link.idx)

        # point_path: link_path를 순서대로 이어붙이기
        point_path = []
        for i, link_id in enumerate(link_path):
            link_obj = self.links[link_id]
            pts = link_obj.points  # [(x,y,z), (x,y,z), ...]
            # 중복 제거: 첫 링크는 모두 추가, 이후 링크는 첫 점을 빼고 추가
            if i == 0:
                for p in pts:
                    point_path.append([p[0], p[1], p[2]])
            else:
                for p_idx, p in enumerate(pts):
                    if p_idx == 0:
                        continue
                    point_path.append([p[0], p[1], p[2]])

        return {
            'node_path': node_path,
            'link_path': link_path,
            'point_path': point_path
        }

    def heuristic(self, node_a, node_b):
        """
        휴리스틱 함수: 단순 유클리드 거리(2D)
        node.point = (x, y)
        """
        ax, ay = node_a.point[0], node_a.point[1]
        bx, by = node_b.point[0], node_b.point[1]
        return sqrt((bx - ax)**2 + (by - ay)**2)

        # 결과적으로 self.node_set.nodes[node_id] = new_node 형태가 됨

    def find_closest_link_and_point(self, pos):
        """ pos=(x,y)와 가장 가까운 link + link.points 상의 포인트 """
        x, y = pos
        min_dist = float('inf')
        closest_link, closest_point = None, None
        for ln in self.line_set.lines.values():
            for pt in ln.points:
                dx = x - pt[0]
                dy = y - pt[1]
                dist = dx*dx + dy*dy
                if dist < min_dist:
                    min_dist = dist
                    closest_link = ln
                    closest_point = pt
        return closest_link, closest_point

    # --------------------------------------------------------------
    # 링크 분할 -> (linkA, linkB) 생성
    # --------------------------------------------------------------
    def split_link_at_point(self, link, new_node):
        """ 
        link.points를 new_node.point 지점에서 둘로 나눠
        linkA, linkB를 생성해 리스트로 반환.
        실제 line_set에 등록/제거는 바깥에서 처리.
        """
        old_points = link.points  # np.array shape=(N,3)

        # 1) split 인덱스 찾기
        split_idx = -1
        for i in range(len(old_points)):
            # np.allclose()로 좌표 비교
            if np.allclose(old_points[i], new_node.point, atol=1e-7):
                split_idx = i
                break
        if split_idx < 0:
            rospy.logwarn("[WARN] split_point not found in link=%s", link.idx)
            # 분할 불가 -> 그저 원본 링크 그대로?
            return []

        # 2) np.split
        pA = old_points[:split_idx+1]
        pB = old_points[split_idx:]

        # 3) linkA 생성
        linkA_id = str(link.idx) + "_A"
        linkA = link.__class__(None, linkA_id)  # Line(None, 'LNxxx') 식
        linkA.set_points(pA)

        linkA.set_from_node(link.from_node)  # 원본 from_node
        linkA.set_to_node(new_node)

        # 3.1) from_node ~ linkA 연결
        if link.from_node and (link in link.from_node.to_links):
            link.from_node.to_links.remove(link)
            link.from_node.to_links.append(linkA)
        new_node.from_links.append(linkA)

        # 4) linkB 생성
        linkB_id = str(link.idx) + "_B"
        linkB = link.__class__(None, linkB_id)
        linkB.set_points(pB)

        linkB.set_from_node(new_node)
        linkB.set_to_node(link.to_node)

        if link.to_node and (link in link.to_node.from_links):
            link.to_node.from_links.remove(link)
            link.to_node.from_links.append(linkB)
        new_node.to_links.append(linkB)

        # 필요하면 cost나 차선정보 복사
        linkA.calculate_cost()
        linkB.calculate_cost()

        return [linkA, linkB]

# Local Path (지역경로) 는 global Path(전역경로) 에서 차량과 가장 가까운 포인트를 시작으로 만들어 집니다.

# 2. Local Path publisher 선언
# 3. Local Path 의 Size 결정
# 4. 콜백함수에서 처음 메시지가 들어오면 초기 위치를 저장

# 7. Local Path 메세지 Publish

class local_path_pub:
    def __init__(self):

        # TODO: (1) Global Path 와 Odometry 데이터 subscriber 생성
        rospy.Subscriber('/odom', Odometry, self.odom_callback)
        rospy.Subscriber('/global_path', Path, self.global_path_callback)

        self.local_path_pub = rospy.Publisher('/local_path', Path, queue_size=1)

        # 초기화
        self.is_odom = False
        self.is_path = False

        # TODO: (3) Local Path 의 Size 결정
        self.local_path_size = 50  # Local Path의 크기 (적절한 범위: 50 ~ 200)

        rate = rospy.Rate(30)  # 20Hz
        while not rospy.is_shutdown():
            if self.is_odom and self.is_path:
                local_path_msg = Path()
                local_path_msg.header.frame_id = 'map'

                x = self.x
                y = self.y

                # TODO: (5) Global Path 에서 차량 위치와 가장 가까운 포인트(Current Waypoint) 탐색
                min_dis = float('inf')
                current_waypoint = -1
                for i, waypoint in enumerate(self.global_path_msg.poses):
                    distance = sqrt(pow(x - waypoint.pose.position.x, 2) + pow(y - waypoint.pose.position.y, 2))
                    if distance < min_dis:
                        min_dis = distance
                        current_waypoint = i

                # TODO: (6) 가장 가까운 포인트(Current Waypoint) 위치부터 Local Path 생성 및 예외 처리
                if current_waypoint != -1:

                        for i in range(current_waypoint, current_waypoint + self.local_path_size):
                            local_path_msg.poses.append(self.global_path_msg.poses[i])
                    else:
                        for i in range(current_waypoint, len(self.global_path_msg.poses)):
                            local_path_msg.poses.append(self.global_path_msg.poses[i])

                # TODO: (7) Local Path 메세지 Publish
                self.local_path_pub.publish(local_path_msg)

            rate.sleep()

    def callback_goal(self, ch, method, properties, body):
        """
        Destination message with optional ETA.
        Response: "lat|lon" or "lat|lon|ETA"
        Request: "destination|lat|lon|ETA"
        """
        try:
            # 1) Decode body
            msg = body.decode("utf-8")
            rospy.loginfo("[Goal] Received message: %s" % msg)

            # 2) Parse
            tokens = msg.strip().split('|')
            if len(tokens) not in [3, 4]:
                rospy.logwarn("[Goal] Invalid message format: %s" % msg)
                return

            lat_str, lon_str = tokens[1], tokens[2]
            lat = float(lat_str)
            lon = float(lon_str)
            eta_given = (len(tokens) == 4)

            # 3) Convert & plan
            goal_xy = self.convertLL2UTM(lon, lat)
            planner = GlobalPathCalculator()
            dist_m, time_s, path_pts = planner.compute_route_time(
                self.current_pos, goal_xy, speed_kmh=30.0
            )
            rospy.loginfo("[Goal] Distance: %.1fm, travel time: %.1fs" % (dist_m, time_s))

            # 4) Compute ETA
            if eta_given:
                eta_str = tokens[3].strip()           # 공백 제거
                eta_str = eta_str.rstrip("Z")         # Z 제거
                if '.' in eta_str:
                    eta_str = eta_str.split('.')[0]   # 마이크로초 제거
                eta_utc = datetime.strptime(eta_str, '%Y-%m-%dT%H:%M:%S')
            else:
                eta_utc = now_utc + timedelta(seconds=time_s)
                eta_str = eta_utc.strftime('%Y-%m-%dT%H:%M:%SZ')

            # 5) Send RPC response (reply_to)
            response_msg = "%s|%s|%s" % (lat_str, lon_str, eta_str)


            credentials = pika.PlainCredentials('guest', 'guest')
            parameters = pika.ConnectionParameters(

            channel.basic_publish(
                exchange='',
                routing_key=properties.reply_to,
                properties=pika.BasicProperties(correlation_id=properties.correlation_id),
                body=response_msg
            )
            channel.basic_ack(delivery_tag=method.delivery_tag)
            connection.close()

            rospy.loginfo("[Goal] Sent RPC response: %s" % response_msg)

            # 6) Wait until departure if ETA was given
            if eta_given:
                seconds_until_departure = (eta_utc - now_utc).total_seconds() - time_s
                if seconds_until_departure > 10:
                    rospy.loginfo("[Goal] Waiting %d seconds before departure..." % int(seconds_until_departure))
                    r = rospy.Rate(1)
                    for _ in range(int(seconds_until_departure)):
                        if rospy.is_shutdown():
                            return
                        r.sleep()
                else:
                    rospy.logwarn("[Goal] Already past ETA, start now.")

# -*- coding: utf-8 -*-

import os

# MGeo 관련 import (사용자 환경에 맞게 조정)

class GlobalPathCalculator(object):
    """
    start=(sx, sy), goal=(gx, gy)를 받아
    1) MGeo + A* 경로 탐색
    2) 경로 길이(m)
    3) 30km/h 평균 속도로 주행 시 소요 시간(초)

    """

        mgeo_data_path: MGeo JSON이 있는 경로
                        (None이면 디폴트 경로 사용)
        """

            # 예시: lib/mgeo_data/R_KR_PG_K-City
            mgeo_data_path = os.path.join(current_path, 'lib/mgeo_data/R_KR_PG_K-City')

        self.node_set = self.mgeo_planner_map.node_set
        self.line_set = self.mgeo_planner_map.link_set

        # dict
        self.nodes = self.node_set.nodes
        self.links = self.line_set.lines

        # A* 객체
        self.global_planner = AStar(self.nodes, self.links)

    def compute_route_time(self, start_xy, goal_xy, speed_kmh=30.0):
        """
        주된 함수:
          1) A*로 경로 생성
          2) 경로 길이 계산
          3) speed_kmh(기본 30km/h)로 달릴 때 걸리는 시간(초) 반환
        
        return: (total_length_m, travel_time_s, path_points)
          - total_length_m: 경로 길이 (m)
          - travel_time_s:  주행 소요 시간 (초)
          - path_points:    (x, y, z) 튜플 리스트
        """
        # (필요 시 매번 MGeo를 리셋하려면 self.reload_mgeo() 등 사용)
        
        node_id_start = "start_node"
        node_id_end   = "end_node"

        start_link, start_closest_pt = self.find_closest_link_and_point(start_xy)
        end_link,   end_closest_pt   = self.find_closest_link_and_point(goal_xy)

        node_start = self.create_node_with_id(node_id_start, start_closest_pt)
        node_end   = self.create_node_with_id(node_id_end,   end_closest_pt)

        # 기존 링크 분할
        new_links_for_start = self.split_link_at_point(start_link, node_start)
        self.line_set.remove_line(start_link)
        for ln in new_links_for_start:
            self.line_set.append_line(ln)

        new_links_for_end = self.split_link_at_point(end_link, node_end)
        self.line_set.remove_line(end_link)
        for ln in new_links_for_end:
            self.line_set.append_line(ln)

        # A* 재생성
        self.nodes = self.node_set.nodes  # 업데이트
        self.links = self.line_set.lines
        self.global_planner = AStar(self.nodes, self.links)

        result, path = self.global_planner.find_shortest_path(node_id_start, node_id_end)
        if not result:
            print("[WARN] A* failed to find path from {} to {}".format(node_id_start, node_id_end))
            return 0.0, 0.0, []

        point_path = path['point_path']  # [(x,y,z), (x,y,z), ...]

        total_length = self.compute_path_length(point_path)

        # km/h -> m/s
        speed_m_s = speed_kmh * 1000.0 / 3600.0
        travel_time_s = total_length / speed_m_s

        return total_length, travel_time_s, point_path

    def compute_path_length(self, path_points):
        """
        path_points: [(x1,y1,z1), (x2,y2,z2), ...]
        """
        total_dist = 0.0
        for i in range(1, len(path_points)):
            x1, y1, z1 = path_points[i-1]
            x2, y2, z2 = path_points[i]
            dx = x2 - x1
            dy = y2 - y1
            dz = z2 - z1
            dist = math.sqrt(dx*dx + dy*dy + dz*dz)
            total_dist += dist
        return total_dist

            for pt in ln.points:
                dx = x - pt[0]
                dy = y - pt[1]
                dist_sq = dx*dx + dy*dy
                if dist_sq < min_dist:
                    min_dist = dist_sq
                    closest_link = ln
                    closest_point = pt
        return closest_link, closest_point

            link.to_node.from_links.append(linkB)

#!/usr/bin/env python
# -*- coding: utf-8 -*-

class pidControl:
    def __init__(self, p_gain=0.3, i_gain=0.00, d_gain=0.03, control_time=0.02):
        self.p_gain = p_gain
        self.i_gain = i_gain
        self.d_gain = d_gain
        self.controlTime = control_time
        self.prev_error = 0
        self.i_control = 0

    def pid(self, target_vel, current_vel):
        error = target_vel - current_vel
        p_control = self.p_gain * error
        self.i_control += error * self.controlTime
        i_term = self.i_gain * self.i_control
        d_term = self.d_gain * (error - self.prev_error) / self.controlTime
        output = p_control + i_term + d_term
        self.prev_error = error
        return output

#!/usr/bin/env python
# -*- coding: utf-8 -*-

import math
import numpy as np
from shapely.geometry import Point, Polygon


class velocityPlanning:
    def __init__(self, car_max_speed, road_friciton):
        self.car_max_speed = car_max_speed
        self.road_friction = road_friciton
        self.school_zones = [
            Polygon([
                (95, 1141), (129, 1203), (123, 1207), (109, 1183),
                (80, 1200), (73, 1189), (102, 1170), (89, 1145)
            ])
        ]

    def curvedBaseVelocity(self, global_path, point_num):
        out_vel_plan = []

        # (초기 point_num 구간 일정속도)
        # for i in range(point_num):
        #     out_vel_plan.append(10)

        # (곡률 기반 계산)
        for i in range(0, len(global_path.poses) - point_num):
            x_list = []
            y_list = []
            for box in range(-point_num, point_num):
                x = global_path.poses[i + box].pose.position.x
                y = global_path.poses[i + box].pose.position.y
                x_list.append([-2 * x, -2 * y, 1])
                y_list.append(-(x ** 2 + y ** 2))
            A = np.array(x_list)
            B = np.array(y_list)
            X = np.linalg.lstsq(A, B, rcond=None)[0]
            r = np.sqrt(X[0] ** 2 + X[1] ** 2 - X[2])

            # v_max = sqrt(r*g*마찰)
            v_max = math.sqrt(r * 9.81 * self.road_friction)
            if v_max > self.car_max_speed:
                v_max = self.car_max_speed
                
            # school_zone limit     
            path_point = global_path.poses[i].pose.position
            point = Point(path_point.x, path_point.y)

            for zone in self.school_zones:
                if zone.contains(point):
                    v_max = min(v_max, 8.33333)  # 제한속도 적용
                
            out_vel_plan.append(v_max)

        # 마지막 일부
        for i in range(len(global_path.poses) - point_num, len(global_path.poses) - 10):
            out_vel_plan.append(8.3333)
        for i in range(len(global_path.poses) -10, len(global_path.poses)):
            out_vel_plan.append(4)
        return out_vel_plan

        # 새로운 글로벌 경로 수신 시, 경로 업데이트 및 (출발) 상태 초기화
        self.global_path = msg.poses
        rospy.loginfo("새로운 글로벌 경로 수신됨")
        # 새로운 경로가 수신되면 출발 상태(예: 0)로 초기화되었다고 가정

    def arrival_status_callback(self, msg):
        # arrive detector에서 발행한 도착 상태 업데이트 (0: 출발, 1: 주행중, 2: 도착)
        self.arrival_state = msg.data
        rospy.loginfo("도착 상태 업데이트: %d", self.arrival_state)

    def odom_callback(self, msg):
        # odom 토픽에서 현재 위치 및 orientation 업데이트
        orientation_q = msg.pose.pose.orientation
        quaternion = (orientation_q.x,
                      orientation_q.y,
                      orientation_q.z,
                      orientation_q.w)
        _, _, self.vehicle_yaw = euler_from_quaternion(quaternion)
        
        self.current_pose.position.x = msg.pose.pose.position.x
        self.current_pose.position.y = msg.pose.pose.position.y
        
        # arrive detector의 상태가 2(도착)이면 방향 계산을 중단
        if self.arrival_state == 2:
            rospy.loginfo("도착 상태(2) - 방향 계산 중단, 새로운 글로벌 경로 수신을 대기")
            return
        
        self.detect_direction()

    def get_current_waypoint(self, ego_xy, global_path):
        """
        ego_xy : [x, y] 형태의 리스트
        global_path : PoseStamped 리스트
        현재 차량 위치와 가장 가까운 waypoint의 인덱스를 반환
        """
        min_dist = float('inf')
        current_waypoint = -1
        for i, pose in enumerate(global_path):
            dx = ego_xy[0] - pose.pose.position.x
            dy = ego_xy[1] - pose.pose.position.y
            dist = math.sqrt(dx * dx + dy * dy)
            if dist < min_dist:
                min_dist = dist
                current_waypoint = i
        return current_waypoint

    def detect_direction(self):
        if not self.global_path or self.current_pose is None:
            return

        # 현재 차량 위치 추출 (ego 좌표)
        ego_xy = [self.current_pose.position.x, self.current_pose.position.y]
        current_idx = self.get_current_waypoint(ego_xy, self.global_path)

        if current_idx == -1:
            rospy.logwarn("현재 위치에 해당하는 waypoint를 찾을 수 없음")
            return

        lookahead = 75  # 기본 lookahead 포인트 개수
        
        # 남은 경로가 부족하면 마지막 포인트 사용
        if current_idx + lookahead >= len(self.global_path):
            future_idx = len(self.global_path) - 1
        else:
            future_idx = current_idx + lookahead

        # 현재 waypoint와 미래 waypoint 사이의 경로 각도 계산
        current_point = self.global_path[current_idx].pose.position
        future_point = self.global_path[future_idx].pose.position
        
        path_heading = math.atan2(future_point.y - current_point.y,
                                  future_point.x - current_point.x)

        # 차량의 yaw와 경로 각도의 차이 계산 (정규화)
        angle_diff = self.normalize_angle(path_heading - self.vehicle_yaw)
        angle_diff_deg = math.degrees(abs(angle_diff))

        # 임계치(10° 이하)는 직진으로 판단
        threshold_deg = 10.0
        if angle_diff_deg < threshold_deg:
            direction = 0
        else:
            # 각도의 부호에 따라 좌회전(양수) 또는 우회전(음수) 결정
            if angle_diff > 0:
                direction = -1
            else:
                direction = 1

        # 결과를 토픽으로 퍼블리시
        msg = Int32()
        msg.data = direction
        self.direction_pub.publish(msg)

        rospy.loginfo("[DirectionDetector] 진행 방향: %d, 차량 각도: %.2f°, 경로 각도: %.2f°, 차이: %.2f°",
                      direction, math.degrees(self.vehicle_yaw), math.degrees(path_heading), angle_diff_deg)

    def normalize_angle(self, angle):
        """각도를 [-pi, pi] 범위로 정규화"""
        while angle > math.pi:
            angle -= 2 * math.pi
        while angle < -math.pi:
            angle += 2 * math.pi
        return angle

if __name__ == "__main__":
    DirectionDetector()
    rospy.spin()
